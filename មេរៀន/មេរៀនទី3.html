៣.១.១ Indexed Array (អារ៉េតាមលំដាប់សន្ទស្សន៍) C1

ក. និយមន័យ (Definition)៖
Indexed Array គឺជាសំណុំទិន្នន័យដែលរក្សាទុកធាតុនីមួយៗដោយប្រើប្រាស់សន្ទស្សន៍ជាលេខ (Numeric Index) ស្វ័យប្រវត្តិ ចាប់ផ្តើមរាប់ពី 0, 1, 2... បន្តបន្ទាប់។
ខ. ទម្រង់ទូទៅ (General Syntax)៖
$array_name = [item0, item1, item2, ...]; $first_item = $array_name[0]; // ចូលប្រើប្រាស់ធាតុតាមសន្ទស្សន៍លេខ
គ. កូដ PHP ពេញលេញ (Light Code Theme)៖
indexed_array.php
<?php
$languages = ["PHP 8", "JavaScript", "SQL"];
echo "ភាសាទី១៖ " . $languages[0] . " | ភាសាទី៣៖ " . $languages[2];
?>
Output
ភាសាទី១៖ PHP 8 | ភាសាទី៣៖ SQL

៣.១.២ Associative Array (អារ៉េតាមគូសោ-តម្លៃ) C2

ក. និយមន័យ (Definition)៖
Associative Array គឺជាអារ៉េដែលប្រើប្រាស់ Named Keys (String) ភ្ជាប់ទៅកាន់តម្លៃ (Key => Value Pair) ដែលងាយស្រួលតំណាងឱ្យព័ត៌មានជាក់លាក់ (ឧទាហរណ៍៖ Student Record)។
ខ. ទម្រង់ទូទៅ (General Syntax)៖
$array_name = [ "key1" => value1, "key2" => value2 ]; $value = $array_name["key1"]; // ចូលប្រើប្រាស់ធាតុតាមឈ្មោះ Key
គ. កូដ PHP ពេញលេញ (Light Code Theme)៖
assoc_array.php
<?php
$student = [
    "id" => "AU-CS-001",
    "name" => "សុខ ជា",
    "gpa" => 3.85
];
echo "អត្តលេខ៖ {$student['id']} | ឈ្មោះ៖ {$student['name']} | GPA: {$student['gpa']}";
?>
Output
អត្តលេខ៖ AU-CS-001 | ឈ្មោះ៖ សុខ ជា | GPA: 3.85

៣.១.៣ Multidimensional Array (អារ៉េពហុវិមាត្រ) C2

ក. និយមន័យ (Definition)៖
Multidimensional Array គឺជាអារ៉េដែលមានផ្ទុកអារ៉េមួយទៀត ឬច្រើននៅខាងក្នុងវា (Array of Arrays) ប្រើប្រាស់សម្រាប់តំណាងឱ្យទិន្នន័យតារាង (Database Result Sets)។
ខ. ទម្រង់ទូទៅ (General Syntax)៖
$matrix = [ ["key1" => val1, "key2" => val2], ["key1" => val3, "key2" => val4] ]; $val = $matrix[row_index]["key_name"];
គ. កូដ PHP ពេញលេញ (Light Code Theme)៖
multi_array.php
<?php
$products = [
    ["title" => "Laptop Dell", "price" => 850],

];
echo "ទំនិញទី១៖ " . $products[0]["title"] . " - តម្លៃ៖ " . $products[0]["price"] . " $";
?>
Output
ទំនិញទី១៖ Laptop Dell - តម្លៃ៖ 850 $

៣.២.១ អនុគមន៍ Built-in សម្រាប់ Array, String & Inspection C2

ក. និយមន័យ (Definition)៖
Built-in Functions គឺជាអនុគមន៍ដែលមានស្រាប់ក្នុង PHP Core ដែលផ្តល់ភាពងាយស្រួលក្នុងការចាត់ចែងទិន្នន័យដោយមិនបាច់សរសេរ Logic ឡើងវិញឡើយ។
ខ. ទម្រង់ទូទៅ (General Syntax)៖
count($array); // រាប់ចំនួនធាតុក្នុង Array array_push($array, $val); // បន្ថែមធាតុទៅចុង Array in_array($val, $array); // ត្រួតពិនិត្យវត្តមានធាតុ (Returns bool) strlen($string); // រាប់ប្រវែងអក្សរ String var_dump($variable); // បង្ហាញ Data Type, Length & Value

builtin_functions.php

$fruits = ["Apple", "Banana"];

ជាអនុគមន៍ដែលគ្រាន់តែប្រតិបត្តិកូដមួយចំនួន (ដូចជាបង្ហាញ UI) ដោយមិនទទួលប៉ារ៉ាម៉ែត្រពីខាងក្រៅ និងមិនបញ្ជូនតម្លៃ return ចេញមកក្រៅវិញឡើយ។

ជាអនុគមន៍ដែលមិនទទួលប៉ារ៉ាម៉ែត្រពីខាងក្រៅឡើយ ប៉ុន្តែវាទាញយក ឬគណនាតម្លៃអ្វីមួយរួចប្រគល់តម្លៃ return ចេញមកក្រៅវិញ។
ទម្រង់ទូទៅ (General Syntax)៖
function functionName(): returnType { // Computation logic return $resultValue; }
កូដ PHP ពេញលេញ (Light Code Theme)៖
func_return_noparam.php
<?php
// 3. Function Return Value - Without Parameter
function getAcademicYear(): string {
    $startYear = 2025;
    $endYear = 2026;
    return "ឆ្នាំសិក្សា {$startYear}-{$endYear}";
}

$currentAcademicYear = getAcademicYear();
echo "ព័ត៌មាន៖ " . $currentAcademicYear;

  • Strict Types (declare(strict_types=1);)៖ បង្ខំឱ្យប្រព័ន្ធផ្ទៀងផ្ទាត់ Data Type នៃ Parameters និង Return Value យ៉ាងតឹងរ៉ឹងបំផុត។

  • Arrow Functions (fn() => expression)៖ ទម្រង់សរសេរ Function កាត់ខ្លីក្នុង PHP 8 ដែលចាប់យក Outer Variables ស្វ័យប្រវត្តិ។

Modular Architecture គឺជាការបំបែកកូដជាចំណែកៗ (Header, Footer, Navigation, Config, Business Logic) ទៅក្នុងឯកសារដាច់ដោយឡែកពីគ្នា ដើម្បីភាពងាយស្រួលថែទាំ និងជៀសវាងការសរសេរកូដដដែលៗ (គោលការណ៍ **DRY - Don't Repeat Yourself**)។ PHP មានបញ្ជា ៤ សម្រាប់តភ្ជាប់ឯកសារ៖
  • include(): បើកឯកសារមកប្រើ។ ប្រសិនបើបាត់ឯកសារ វានឹងបោះតែ Warning ហើយ **បន្តប្រកាសប្រតិបត្តិកូដខាងក្រោមទៀត**។
  • require(): បើកឯកសារមកប្រើ។ ប្រសិនបើបាត់ឯកសារ វានឹងបោះ Fatal Error ហើយ **បញ្ឈប់កម្មវិធីភ្លាមៗ**។
  • include_once(): បើករកឯកសារ។ ប្រសិនបើឯកសារធ្លាប់ត្រូវបាន Load រួចហើយ វានឹងមិន Load ជាន់គ្នាជាលើកទី២ ឡើយ (Warning បើបាត់)។
  • require_once(): បើករកឯកសារ។ ធានាថា Load តែ **១ ដងគត់** និងបោះ **Fatal Error** ឈប់រត់ភ្លាមបើបាត់ឯកសារ។
