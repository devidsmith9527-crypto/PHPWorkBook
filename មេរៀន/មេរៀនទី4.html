៤.១.១ HTTP GET Method & $_GET Superglobal C1

ក. និយមន័យ (Definition)៖
HTTP GET Method គឺជាវិធីសាស្ត្រផ្ញើទិន្នន័យពី Client ទៅកាន់ Server ដោយបង្ហាញទិន្នន័យចំហរនៅលើ URL Query String (ឧទាហរណ៍៖ search.php?keyword=PHP8&page=2)។ ទិន្នន័យត្រូវបានរក្សាទុកក្នុង Associative Array ឈ្មោះថា $_GET។ វិធីសាស្ត្រនេះស័ក្តិសមសម្រាប់ Form ស្វែងរក (Search), Pagination, និងការ Filter ទិន្នន័យ ដែលអនុញ្ញាតឱ្យ User អាច Bookmark ទុក URL បាន។
ខ. ទម្រង់ទូទៅ (General Syntax)៖
<!-- HTML Form --> <form method="GET" action="target.php"> <input type="text" name="param_name"> </form> // PHP Code Accessing GET Parameter $value = $_GET['param_name'] ?? 'default_value';
គ. កូដ PHP ពេញលេញ (Full PHP Code)៖
get_method_demo.php
<?php
// ចាប់យកទិន្នន័យពី GET Query String ដោយប្រើ Null Coalescing Operator (??)
$keyword = $_GET['q'] ?? '';
$category = $_GET['cat'] ?? 'all';
?>
<form method="GET" action="get_method_demo.php">
    <input type="text" name="q" value="<?= htmlspecialchars($keyword) ?>" placeholder="ស្វែងរក...">
    <button type="submit">ស្វែងរក</button>
</form>

<?php if (!empty($keyword)): ?>
    <p>លទ្ធផលស្វែងរកសម្រាប់៖ <b><?= htmlspecialchars($keyword) ?></b> (ប្រភេទ៖ <?= htmlspecialchars($category) ?>)</p>
<?php endif; ?>

គ. កូដ PHP ពេញលេញ (Full PHP Code)៖

<?php
$errors = [];
$fullname = '';

if ($_SERVER["REQUEST_METHOD"] === "POST") {
    // 1. Clean trim leading/trailing spaces
    $fullname = trim($_POST['fullname'] ?? '');

    // 2. Check for empty values
    if (empty($fullname)) {

• ដំណោះស្រាយ Lab 4.9 (Full Secure Form Application) C6៖
secure_register_app.php
<?php
declare(strict_types=1);
$errors = []; $name = ''; $email = '';

if ($_SERVER["REQUEST_METHOD"] === "POST") {
    $name = trim($_POST['name'] ?? '');
    $email = trim($_POST['email'] ?? '');

    if (empty($name)) { $errors['name'] = "សូមបញ្ចូលឈ្មោះ!"; }
    if (!filter_var($email, FILTER_VALIDATE_EMAIL)) { $errors['email'] = "អ៊ីមែលមិនត្រឹមត្រូវ!"; }

    if (empty($errors)) {
        header("Location: secure_register_app.php?success=1");
        exit();
    }
}
?>
<form method="POST">
    <input type="text" name="name" value="<?= htmlspecialchars($name) ?>">
    <span style="color:red;"><?= $errors['name'] ?? '' ?></span><br>
    
    <input type="email" name="email" value="<?= htmlspecialchars($email) ?>">
    <span style="color:red;"><?= $errors['email'] ?? '' ?></span><br>
    <button type="submit">Register</button>
</form>
